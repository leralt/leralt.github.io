一个简单题，做出来比较容易，但代码的简洁程度有所区别

```cpp
void moveZeroes(vector<int>& nums) {
    int n = nums.size(), left = 0, right = 0;
    while (right < n) {
        if (nums[right]) { //判断右边是否不为0，找到右边不为0的第一个数，比较简洁
            swap(nums[left], nums[right]);  // 直接swap而不是分两步
            left++;
        }
        right++;
    }
}
```
